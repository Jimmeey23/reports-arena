# ------------------------------------------------------------ round-5 additions: the controlled comparison
_S = D.XB[D.XB['dt'] >= '2026-01-01'].copy()
_S['capr'] = _S['Capacity'] / _S['Classes'].clip(lower=1)
_S['slot'] = _S['Class'].astype(str) + ' | ' + _S['Day'].astype(str) + ' | ' + _S['Time'].astype(str)
_G = _S.groupby('slot').agg(n=('SessionID', 'size'), trainers=('Trainer', 'nunique'))
_rows = []
for _sl, _r in _G[_G['trainers'] >= 2].iterrows():
    _sub = _S[_S['slot'] == _sl]
    _t = _sub.groupby('Trainer').agg(n=('SessionID', 'size'), att=('CheckedIn', 'mean'), rev=('Revenue', 'mean'))
    _t = _t[_t['n'] >= 3]
    if len(_t) < 2:
        continue
    _t = _t.sort_values('att', ascending=False)
    _rows.append(dict(slot=_sl, best=_t.index[0], best_att=float(_t['att'].iloc[0]), worst=_t.index[-1],
                      worst_att=float(_t['att'].iloc[-1]), spread=float(_t['att'].iloc[0] - _t['att'].iloc[-1]),
                      spread_pct=float((_t['att'].iloc[0] - _t['att'].iloc[-1]) / max(_t['att'].iloc[-1], .01) * 100),
                      n_trainers=len(_t)))
CSLOT = pd.DataFrame(_rows).sort_values('spread', ascending=False)
CS_N = len(CSLOT)
CS_BEST = float(CSLOT['best_att'].mean()); CS_WORST = float(CSLOT['worst_att'].mean())
CS_MED = float(CSLOT['spread'].median()); CS_BIG = int((CSLOT['spread'] > 4).sum())
CS_PCT = CS_BIG / max(CS_N, 1) * 100

def _slotlab(sl):
    _p = sl.split(' | ')
    return f"{_p[0].replace('Studio ','')} \\u2014 {_p[1]} {str(_p[2])[:5]}"

    note=f'Each bar is one slot in which two or more trainers have each taught at least three times during 2026. '
         f'Only {CS_BIG} of {CS_N} slots show a gap wider than four people.')
